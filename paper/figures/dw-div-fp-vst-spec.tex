Definition dw_div_fp_spec :=
  DECLARE _dw_div_fp
  WITH z_ptr : val, x_ptr : val, x_pair : ftype Tdouble * ftype Tdouble, y : ftype Tdouble
  PRE [tptr dword, tptr dword, tdouble]
    PROP ()
    PARAMS (z_ptr; x_ptr; Vfloat y)
    SEP (data_at_ Tsh dword z_ptr; data_at Tsh dword (f2val x_pair) x_ptr)
  POST [tint]
    EX status : int,
      PROP (default_wrapper_status status;
            status = Int.zero -> DWDivFP_guarantee (fst x_pair) (snd x_pair) y)
      RETURN (Vint status)
      SEP ((if Int.eq status Int.zero
            then data_at Tsh dword (f2val (DWDivFP (fst x_pair) (snd x_pair) y)) z_ptr
            else data_at_ Tsh dword z_ptr);
           data_at Tsh dword (f2val x_pair) x_ptr).
